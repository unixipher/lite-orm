struct User {
  id        Int32    @id @default(autoincrement())
  email     String   @unique
  name      String
  active    Boolean  @default(true)
  createdAt DateTime @default(now()) @map("created_at")
  posts     Post[]
}

struct Post {
  id        Int32    @id @default(autoincrement())
  title     String
  content   String?
  userId    Int32    @map("user_id")
  user      User     @relation(fields: [userId], references: [id])
}
mod liteorm;
use lite_orm::{Database, Model};
use liteorm::{User, Post};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let db = Database::connect_env().await?;

    // Type-safe SELECT with parameterized WHERE
    let active_users = db
        .all(User::select()
            .where_(User::ACTIVE.equals(true))
            .limit(20))
        .await?;

    // Safe deletion guarded against unconstrained drops
    db.delete(Post::delete().where_(Post::USER_ID.equals(12))).await?;

    Ok(())
}

CREATE TABLE "User" (
  "id" SERIAL PRIMARY KEY,
  "email" TEXT NOT NULL UNIQUE,
  "name" TEXT NOT NULL,
  "active" BOOLEAN NOT NULL DEFAULT true,
  "created_at" TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE "Post" (
  "id" SERIAL PRIMARY KEY,
  "title" TEXT NOT NULL,
  "content" TEXT,
  "user_id" INTEGER NOT NULL REFERENCES "User"("id")
);

COMMIT;

Engineered for Rust Reliability

Prisma's developer ergonomics fused with Rust's uncompromising safety and performance.

SAFETY

Guarded Mutations

Destructive mutations like UPDATE and DELETE strictly require filter predicates or an explicit override, preventing accidental database wipes.

TRANSACTIONS

Atomic Migrations

Every generated migration and its audit log execute inside a single PostgreSQL transaction with schema-scoped session advisory locks for zero race conditions.

DESKTOP

Native Studio

100% native desktop management app built with Dioxus Desktop. No web server, zero JS bloat, and instantaneous live relation inspections.

INTROSPECTION

Bi-Directional Sync

Run liteorm db pull to reverse-engineer existing PostgreSQL tables into schemas, or liteorm db push to apply additive changes instantly.

SQL

Safe Parameterization

All query values use PostgreSQL bound parameters and never string concatenation. Full support for platform TLS via rustls.

Get Started in Seconds

Follow the standard schema-first workflow from zero to production.

1

Initialize project and schema

Generate a starter liteorm.schema and .env file with database connection string.

liteorm init
2

Sync with PostgreSQL

Directly push additive changes to PostgreSQL during rapid prototyping, or create versioned migration files.

liteorm db push # Direct additive sync
liteorm migrate dev # Versioned transactional migration
3

Generate Rust models and launch Studio

Compile schema definitions into strictly typed Rust structs and launch the cross-platform desktop UI.

liteorm generate # Generates src/liteorm.rs
liteorm studio # Launches native Desktop Studio
